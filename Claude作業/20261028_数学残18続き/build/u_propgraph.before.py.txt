"""特訓：比例のグラフ（教科書 p126〜129）。"""
from fractions import Fraction as F
from mk import Unit
from vutil import only, pt

U = Unit('propgraph', '教科書 p126〜129（比例のグラフ）',
         '2026-10-08 Claude作成（特訓・下書き）。教科書 p126〜129 の比例のグラフを、2つのグラフと図形・グラフ上の格子点・傾きのくらべ方などに広げた一段難しい問題。'
         '答えは check.py で、座標を使った計算と総当たりで検算。'
         '【保留】比例のグラフで、教科書問題（通る点・式を読む）と似ない考え方の問題は 12 問まで。70問にしていない。')
A = [F(n, 60) for n in range(-1200, 1201)]
def area(p, q, r):
    return abs((q[0] - p[0]) * (r[1] - p[1]) - (r[0] - p[0]) * (q[1] - p[1])) / F(2)

U.num('y ＝ −2x のグラフと y ＝ x/2 のグラフの上で、x 座標がどちらも 4 である2点の間のきょりを求めよう（1目もり 1 cm）。', '10',
      '①y ＝ −2x 上は (4, −8)、y ＝ x/2 上は (4, 2)　②同じ x 座標なので、きょりは y 座標の差　③2 − (−8) ＝ 10。答え 10cm',
      lambda: abs(F(4, 2) - (-2 * 4)), units=['cm'])
U.num('y ＝ 3x のグラフ、y ＝ −x のグラフ、x 座標が 2 の点を結んだ直線（y 軸に平行）で囲まれた三角形の面積を求めよう（1目もり 1 cm）。', '8',
      '①頂点は原点、(2, 6)、(2, −2)　②底辺（縦の辺）6 − (−2) ＝ 8、高さ 2　③8 × 2 ÷ 2 ＝ 8。答え 8cm²',
      lambda: area((0, 0), (2, 6), (2, -2)), units=['cm²'])
U.num('2点 A(2, 1)、B(2, 5) を結ぶ線分がある。y ＝ ax のグラフがこの線分と交わるような a のうち、もっとも大きい値を求めよう。', '5/2',
      '①y ＝ ax は x ＝ 2 で y ＝ 2a　②線分と交わるのは 1 ≦ 2a ≦ 5　③a の最大は 5/2（B を通るとき）。答え 5/2',
      lambda: max(a for a in A if 1 <= 2 * a <= 5))
U.num('比例のグラフで、x が −1 から 3 までふえると、y は 8 へる。比例定数を求めよう。', '-2',
      '①x は 4 ふえる　②4a ＝ −8　③a ＝ −2。答え −2',
      lambda: only(a for a in A if a * 3 - a * -1 == -8))
U.num('y ＝ 2x のグラフ上に点Pがあり、Pの x 座標は正の数 t である。Pから x 軸に垂線をひき、x 軸との交点をHとする。三角形 OPH の面積が 25 のとき、t の値を求めよう。', '5',
      '①P(t, 2t)、H(t, 0)　②面積 t × 2t ÷ 2 ＝ t × t ＝ 25　③t ＝ 5。答え 5',
      lambda: only(t for t in A if t > 0 and area((0, 0), (t, 2 * t), (t, 0)) == 25))
U.num('y ＝ x/3 のグラフ上の点Aと、y ＝ 2x のグラフ上の点Bは、どちらも y 座標が 6 である。AB の長さを求めよう（1目もり 1 cm）。', '15',
      '①A：x/3 ＝ 6 より x ＝ 18　②B：2x ＝ 6 より x ＝ 3　③AB ＝ 18 − 3 ＝ 15。答え 15cm',
      lambda: only(x for x in A if x / 3 == 6) - only(x for x in A if 2 * x == 6), units=['cm'])
U.num('長方形 OABC で、O は原点、A(6, 0)、C(0, 4) である。対角線 OB をふくむ比例のグラフの比例定数を求めよう。', '2/3',
      '①B ＝ (6, 4)　②比例定数 ＝ 4 ÷ 6　③2/3。答え 2/3',
      lambda: F(4, 6))
U.num('y ＝ 2x/3 のグラフ上にある点のうち、x 座標も y 座標も整数で、−10 ≦ x ≦ 10 であるものは何個あるか答えよう。', '7',
      '①y が整数になるのは x が 3 の倍数のとき　②−10〜10 の 3 の倍数：−9、−6、−3、0、3、6、9　③7 個。答え 7個',
      lambda: sum(1 for x in range(-10, 11) if (F(2 * x, 3)).denominator == 1), units=['個'])
U.num('比例のグラフが点 (4, 6) を通る。このグラフは点 (−2, k) と点 (k, m) も通る。m の値を求めよう。', '-9/2',
      '①比例定数 6/4 ＝ 3/2　②k ＝ (3/2) × (−2) ＝ −3　③m ＝ (3/2) × (−3) ＝ −9/2。答え −9/2',
      lambda: (lambda a: a * (a * -2))(F(6, 4)))
U.num('y ＝ −x/2 のグラフ上で、x 座標が y 座標より 6 大きい点の x 座標を求めよう。', '4',
      '①y ＝ −x/2 を x − y ＝ 6 に入れる：x + x/2 ＝ 6　②3x/2 ＝ 6　③x ＝ 4（点 (4, −2)）。答え 4',
      lambda: only(x for x in A if x - (-x / 2) == 6))
U.order('次のア〜エの比例のグラフを、かたむきが急な順（x 軸からはなれている順）に並べて記号で答えよう。\nア y ＝ −3x\nイ y ＝ x/2\nウ y ＝ 2x\nエ y ＝ −x/4', 'アウイエ',
        '①比例定数の絶対値が大きいほど、グラフは急になる　②絶対値：ア 3、イ 1/2、ウ 2、エ 1/4　③急な順に ア、ウ、イ、エ。答え アウイエ',
        lambda: ''.join(n for _, n in sorted(zip([3, F(1, 2), 2, F(1, 4)], 'アイウエ'), reverse=True)))
U.num('点 (3, 6) を通る比例のグラフと、点 (3, −3) を通る比例のグラフがある。x 座標が −2 である2つのグラフ上の点の間のきょりを求めよう（1目もり 1 cm）。', '6',
      '①y ＝ 2x と y ＝ −x　②x ＝ −2 で y ＝ −4 と y ＝ 2　③きょり 2 − (−4) ＝ 6。答え 6cm',
      lambda: abs(F(6, 3) * -2 - F(-3, 3) * -2), units=['cm'])
