from fractions import Fraction as F
from itertools import combinations, product
from mk import Unit
from vutil import only, pt

U = Unit('spaceuse', '教科書 p214〜215（空間図形の利用）',
         '2026-10-08 Claude作成（特訓・下書き）。容器の水・積み木など、立体の考え方を使う一段難しい問題。答えは check.py で計算・数え上げで検算。'
         '【保留】図なしで伝えられ、教科書問題と似ない問題は 3 問まで。70問にしていない。')
U.num('底面の半径 5 cm の円柱の容器に、深さ 8 cm まで水が入っている。この水を全部、底面の半径 10 cm の円柱の容器に移すと、深さは何 cm になるか答えよう。', '2',
      '①水の体積 π × 25 × 8 ＝ 200π　②π × 100 × h ＝ 200π　③h ＝ 2。答え 2cm',
      lambda: F(25 * 8, 100), units=['cm'])
U.num('高さ 12 cm の円錐の形のコップ（頂点が下）に、頂点から高さ 6 cm のところまで水を入れた。水の体積は、コップ全体の容積の何倍か答えよう。', '1/8',
      '①水の部分も円錐で、高さも底面の半径もコップの 1/2　②体積は (1/2) × (1/2) × (1/2) ＝ 1/8 倍　③答え 1/8倍',
      lambda: F(1, 2) ** 3, units=['倍'])
U.num('1辺 1 cm の立方体を 27 個積んで 3 × 3 × 3 の立方体をつくり、外側の面全部にペンキをぬった。ちょうど2つの面がぬられた小さな立方体は何個あるか答えよう。', '12',
      '①2面がぬられるのは、大きい立方体の辺の上でかど以外のもの　②各辺に 1 個、辺は 12 本　③12 個。答え 12個',
      lambda: sum(1 for x, y, z in product(range(3), repeat=3) if sum(v in (0, 2) for v in (x, y, z)) == 2), units=['個'])
