"""特訓：立体の構成（教科書 p200〜201：面や線を動かしてできる立体・展開図）。"""
from fractions import Fraction as F
from itertools import combinations, product
from mk import Unit
from vutil import only, pt

U = Unit('solidmake', '教科書 p200〜201（立体の構成・展開図）',
         '2026-10-08 Claude作成（特訓・下書き）。面や線を動かしてできる立体と展開図を、数で答える一段難しい問題。答えは check.py で計算し直して検算。'
         '【保留】図なしで伝えられ、教科書問題と似ない問題は 4 問まで。70問にしていない。')
U.expr('底面の半径 3 cm、高さ 5 cm の円柱の展開図で、側面の長方形の横の長さ（底面の円のまわりと同じ）を、π を使って答えよう（cm）。', '6π',
       '①側面の長方形の横 ＝ 底面の円周　②2π × 3 ＝ 6π　③答え 6π cm',
       lambda π: 2 * π * 3)
U.num('円錐の展開図で、側面のおうぎ形が半円で、母線の長さが 8 cm である。底面の円の半径を答えよう。', '4',
      '①半円の弧 ＝ 2π × 8 ÷ 2 ＝ 8π　②底面の円周 2πr ＝ 8π　③r ＝ 4。答え 4cm',
      lambda: only(r for r in [F(n, 10) for n in range(0, 200)] if 2 * r == 8), units=['cm'])
U.num('1辺 3 cm の正方形を、その面に垂直な方向に 5 cm 平行に動かしてできる立体の体積を求めよう。', '45',
      '①正方形が動いたあとは、底面が正方形の四角柱（直方体）　②3 × 3 × 5 ＝ 45　③答え 45cm³',
      lambda: 3 * 3 * 5, units=['cm³'])
U.expr('長さ 4 cm の線分を、それと平行で 3 cm はなれた直線を軸として1回転させた。線分が動いたあとにできる面（円柱の側面）の面積を、π を使って答えよう（cm²）。', '24π',
       '①できるのは底面の半径 3 cm、高さ 4 cm の円柱の側面　②2π × 3 × 4　③24π。答え 24π cm²',
       lambda π: 2 * π * 3 * 4)
