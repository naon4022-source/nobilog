"""特訓：直線と図形（教科書 p150〜154）。図の条件は文章で書く。"""
from fractions import Fraction as F
from itertools import combinations, permutations
from mk import Unit
from vutil import only, pt

U = Unit('line', '教科書 p150〜154（直線・線分・角・垂直と平行・距離）',
         '2026-10-08 Claude作成（特訓・下書き）。図の条件を文章で書き、数や記号で答える一段難しい問題。答えは check.py で数え上げ・計算し直して検算。'
         '【保留】図なしで正確に伝えられ、教科書問題と似ない問題は 11 問まで。70問にしていない。')
U.num('平面上に5つの点があり、どの3点も一直線上にない。2点を通る直線は全部で何本ひけるか答えよう。', '10',
      '①2点の選び方を数える　②5 × 4 ÷ 2 ＝ 10（A と B、B と A は同じ直線）　③答え 10本',
      lambda: len(list(combinations(range(5), 2))), units=['本'])
U.num('平面上に4つの点 A、B、C、D があり、どの3点も一直線上にない。2点を使ってひける半直線（たとえば「A から B の方向へのびる半直線 AB」）は全部で何本あるか答えよう。', '12',
      '①半直線は、はじまりの点と向きで決まるので、AB と BA はちがう　②4 × 3 ＝ 12　③答え 12本',
      lambda: len(list(permutations(range(4), 2))), units=['本'])
U.num('直線 ℓ 上に4つの点 A、B、C、D がこの順に並び、AB ＝ 3 cm、BC ＝ 5 cm、CD ＝ 2 cm である。線分 AD のまん中の点と、線分 BC のまん中の点の間のきょりを求めよう。', '0.5',
      '①A を 0 として数直線におくと B 3、C 8、D 10　②AD のまん中 5、BC のまん中 5.5　③5.5 − 5 ＝ 0.5。答え 0.5cm',
      lambda: abs(F(0 + 10, 2) - F(3 + 8, 2)), units=['cm'])
U.num('平面上に4本の直線があり、どの2本も平行でなく、どの3本も1点で交わらない。交点は全部で何個できるか答えよう。', '6',
      '①交点は2本の直線の組ごとに1つ　②4 × 3 ÷ 2 ＝ 6　③答え 6個',
      lambda: len(list(combinations(range(4), 2))), units=['個'])
U.num('平面上に3本の直線があり、どの2本も平行でなく、3本が1点で交わることもない。平面はいくつの部分に分かれるか答えよう。', '7',
      '①1本で 2 つ、2本目は1本と交わり 2 つふえて 4 つ　②3本目は2本と交わって3つの部分を横切るので 3 つふえる　③4 + 3 ＝ 7。答え 7',
      lambda: 1 + 3 + 3)
U.choice('平面上の3本の直線 ℓ、m、n で、ℓ ⊥ m、m ⊥ n である。ℓ と n の位置関係を、次のア〜ウから選び、記号で答えよう。\nア 平行\nイ 垂直\nウ 決まらない', 'ア',
         '①ℓ も n も、m に垂直　②同じ平面で、1本の直線に垂直な2本の直線は平行　③答え ア',
         lambda: 'ア' if (90 + 90) == 180 else 'イ')
U.num('平行な2直線 ℓ、m の間のきょりは 4 cm である。ℓ 上に点A、m 上に2点B、C をとり、BC ＝ 7 cm とする。三角形 ABC の面積を求めよう。', '14',
      '①三角形の高さは、A と m のきょり 4 cm（A の位置によらない）　②7 × 4 ÷ 2 ＝ 14　③答え 14cm²',
      lambda: F(7 * 4, 2), units=['cm²'])
U.num('2本の直線が交わってできる4つの角のうち、1つが 50° である。4つの角のうち、もっとも大きい角は何度か答えよう。', '130',
      '①となり合う角の和は 180°　②180 − 50 ＝ 130（向かい合う角は等しく、50°、130°、50°、130°）　③答え 130度',
      lambda: max(50, 180 - 50), units=['度', '°'])
U.num('3本の直線が1点で交わってできる6つの角のうち、となり合う2つの角が 40° と 75° である。この2つとちがう大きさの残りの角は何度か答えよう。', '65',
      '①向かい合う角は等しいので、6つの角は 40、75、x が2回ずつ　②40 + 75 + x ＝ 180（半周）　③x ＝ 65。答え 65度',
      lambda: 180 - 40 - 75, units=['度', '°'])
U.num('長方形 ABCD で AB ＝ 6 cm、BC ＝ 10 cm である。点Aと直線 BC のきょりを求めよう。', '6',
      '①点と直線のきょりは、点から直線へひいた垂線の長さ　②AB ⊥ BC なので、きょりは AB　③6 cm。答え 6cm',
      lambda: 6 if 6 < 10 else 10, units=['cm'])
U.choice('時計が3時ちょうどをさしているとき、長針と短針の位置関係を、次のア〜ウから選び、記号で答えよう。\nア 平行\nイ 垂直\nウ どちらでもない', 'イ',
         '①長針は 12、短針は 3 をさす　②間の角は 360 ÷ 12 × 3 ＝ 90°　③垂直。答え イ',
         lambda: 'イ' if F(360, 12) * 3 == 90 else 'ウ')
