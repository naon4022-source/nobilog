from fractions import Fraction as F
from itertools import combinations, permutations
from mk import Unit
from vutil import only, pt

U = Unit('draw1', '教科書 p162〜163（垂直二等分線・角の二等分線）',
         '2026-10-08 Claude作成（特訓・下書き）。作図そのものではなく、垂直二等分線・角の二等分線の性質を使って数で答える一段難しい問題。答えは check.py で計算し直して検算。'
         '【保留】作図は図をかく学習で、文章で数を答える形にできる問題は 6 問まで。70問にしていない。')
U.coord('2点 A(−2, 1)、B(6, 1) の垂直二等分線と x 軸の交点の座標を答えよう。', ('2', '0'),
        '①AB のまん中の点は (2, 1)　②AB は x 軸に平行なので、垂直二等分線は x 座標が 2 の直線（y 軸に平行）　③x 軸との交点 (2, 0)。答え (2, 0)',
        lambda: pt(F(-2 + 6, 2), 0))
U.num('2点 A(1, 1)、B(1, 7) の垂直二等分線は、x 軸に平行な直線になる。この直線上の点の y 座標を答えよう。', '4',
      '①AB のまん中の点は (1, 4)　②AB は y 軸に平行なので、垂直二等分線は y 座標 4 の直線　③答え 4',
      lambda: F(1 + 7, 2))
U.num('∠AOB ＝ 70° の角の二等分線をひいた。二等分線と半直線 OA がつくる角は何度か答えよう。', '35',
      '①角の二等分線は角を2等分する　②70 ÷ 2 ＝ 35　③答え 35度',
      lambda: F(70, 2), units=['度', '°'])
U.num('三角形 ABC で ∠B ＝ 60°、∠C ＝ 80° である。∠B と ∠C の二等分線の交点をIとするとき、∠BIC の大きさを求めよう。', '110',
      '①∠IBC ＝ 30°、∠ICB ＝ 40°　②三角形 IBC の内角の和は 180°　③∠BIC ＝ 180 − 30 − 40 ＝ 110。答え 110度',
      lambda: 180 - F(60, 2) - F(80, 2), units=['度', '°'])
U.num('三角形 ABC の3つの辺の垂直二等分線は1点Oで交わる。OA ＝ 5 cm のとき、OB + OC の長さを求めよう。', '10',
      '①O は AB の垂直二等分線上にあるので OA ＝ OB　②BC の垂直二等分線上にもあるので OB ＝ OC　③5 + 5 ＝ 10。答え 10cm',
      lambda: 5 + 5, units=['cm'])
U.pick('定規とコンパスだけの作図（正三角形・垂直二等分線・角の二等分線の組み合わせ）で、直接つくることができる角を、次のア〜エからすべて選び、記号で答えよう。\nア 15°\nイ 20°\nウ 75°\nエ 100°', 'アウ',
       '①正三角形で 60°、二等分をくり返して 30°、15°　②垂線で 90°、60° + 15° ＝ 75° もつくれる　③20°、100° はこの方法ではつくれない（60 の 1/3 は二等分ではつくれない）　④答え ア、ウ',
       lambda: {n for n, a in zip('アイウエ', [15, 20, 75, 100]) if a in {x + y for x in (0, 15, 30, 45, 60, 90) for y in (0, 15, 30, 45, 60, 90)}})
