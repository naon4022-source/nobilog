"""特訓：反比例の式（教科書 p131〜133）。"""
from fractions import Fraction as F
from mk import Unit
from vutil import only

U = Unit('inv1', '教科書 p131〜133（反比例と比例定数）',
         '2026-10-08 Claude作成（特訓・下書き）。教科書 p131〜133 の反比例・比例定数を、変化の量・自然数の組・比例との組み合わせなどに広げた一段難しい問題。'
         '答えは check.py で総当たりにより検算。'
         '【保留】反比例の式の節は短く、教科書問題と似ない考え方の問題は 11 問まで。70問にしていない。')
A = [F(n, 60) for n in range(-3000, 3001)]

U.num('y は x に反比例し、x の値が 4 から 6 にふえると、y の値は 3 へる。比例定数を求めよう。', '36',
      '①y ＝ a/x で a/4 − a/6 ＝ 3　②a/12 ＝ 3　③a ＝ 36。答え 36',
      lambda: only(a for a in A if a / 4 - a / 6 == 3))
U.num('xy ＝ 24 を満たす自然数 x、y の組は何組あるか答えよう。', '8',
      '①x は 24 の約数：1、2、3、4、6、8、12、24　②それぞれに y が1つずつ決まる　③8 組。答え 8組',
      lambda: sum(1 for x in range(1, 25) for y in range(1, 25) if x * y == 24), units=['組'])
U.num('x が 2 から 6 まで変わるとき、反比例 y ＝ 12/x の値が動くはんいの大きさ（いちばん大きい値 − いちばん小さい値）を求めよう。', '4',
      '①x ＞ 0 では、x が大きいほど y は小さい　②x ＝ 2 で 6、x ＝ 6 で 2　③6 − 2 ＝ 4。答え 4',
      lambda: (lambda ys: max(ys) - min(ys))([12 / x for x in A if 2 <= x <= 6]))
U.num('x が −6 以上 −2 以下のどんな値をとっても、反比例 y ＝ −18/x の値はある数をこえない。その数（y のいちばん大きい値）を求めよう。', '9',
      '①x ＝ −6 で y ＝ 3、x ＝ −2 で y ＝ 9　②この範囲では x が大きいほど y も大きい　③最大は 9。答え 9',
      lambda: max(F(-18) / x for x in A if -6 <= x <= -2))
U.choice('y は x に比例し、z は y に反比例する（x、y、z は 0 でない）。z と x の関係として正しいものを、次のア〜ウから選び、記号で答えよう。\nア z は x に比例する\nイ z は x に反比例する\nウ どちらでもない', 'イ',
         '①y ＝ ax、z ＝ b/y とすると z ＝ b/(ax) ＝ (b/a)/x　②z は x に反比例する　③答え イ',
         lambda: 'イ' if len({(F(6) / (2 * x)) * x for x in (F(1), F(3), F(-5))}) == 1 else 'ア')
U.num('y は x に反比例し、さらに z は y に反比例している（どれも 0 でない）。x の値を 3 倍にすると、z の値は何倍になるか答えよう。', '3',
      '①x を 3 倍すると、y は 1/3 倍　②y が 1/3 倍になると、z は 3 倍（反比例）　③z は x に比例していることになる。答え 3倍',
      lambda: only((F(6) / (F(4) / (3 * x))) / (F(6) / (F(4) / x)) for x in (F(1), F(2), F(-5))), units=['倍'])
U.num('面積が 24 cm² の長方形の縦を x cm、横を y cm とする。縦を 1.5 倍にすると、横はもとの何倍になるか答えよう。', '2/3',
      '①xy ＝ 24（y は x に反比例）　②x を 1.5 倍すると、y は 1 ÷ 1.5 ＝ 2/3 倍　③答え 2/3倍',
      lambda: only((F(24) / (x * F(3, 2))) / (F(24) / x) for x in (F(2), F(3), F(8))), units=['倍'])
U.num('y は x に反比例している。x の値を 20% ふやすと、y の値はもとの値のどれだけ（分数で）へるか答えよう。', '1/6',
      '①x を 1.2 倍すると y は 1 ÷ 1.2 ＝ 5/6 倍　②へる量は 1 − 5/6 ＝ 1/6　③答え 1/6',
      lambda: only(1 - (F(12) / (x * F(6, 5))) / (F(12) / x) for x in (F(2), F(5))))
U.num('y ＝ 36/x で、x と y がどちらも自然数で、x ＜ y となる組は何組あるか答えよう。', '4',
      '①xy ＝ 36 の自然数の組：(1, 36)、(2, 18)、(3, 12)、(4, 9)、(6, 6)、…　②x ＜ y は (1, 36)、(2, 18)、(3, 12)、(4, 9)　③4 組。答え 4組',
      lambda: sum(1 for x in range(1, 37) for y in range(1, 37) if x * y == 36 and x < y), units=['組'])
U.num('y は x に反比例する。x の値が 1 ふえると、y の値が 1/2 倍になった。はじめの x の値を求めよう（x は正の数）。', '1',
      '①y ＝ a/x、x + 1 のとき a/(x + 1)　②a/(x + 1) ＝ (1/2) × a/x より x + 1 ＝ 2x　③x ＝ 1。答え 1',
      lambda: only(x for x in A if x > 0 and 12 / (x + 1) == F(1, 2) * (12 / x)))
U.num('y は x に反比例し、x ＝ 3 のとき y ＝ −8 である。x の値と y の値が等しくなるような点はあるか。x と y の値の積を答えよう（その積から考えよう）。', '-24',
      '①比例定数 a ＝ 3 × (−8) ＝ −24 なので、いつも xy ＝ −24　②x と y が等しいと xy は 0 以上になり −24 にならない（等しくなる点はない）　③積は −24。答え −24',
      lambda: only(x * F(-24) / x for x in (F(1), F(-3), F(6))))
