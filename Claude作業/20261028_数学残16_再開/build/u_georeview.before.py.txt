"""特訓：平面図形 ふりかえりドリル（5章全体）。"""
from fractions import Fraction as F
from itertools import combinations, permutations
from mk import Unit
from vutil import only, pt

U = Unit('georeview', '教科書 p150〜176（5章 平面図形 全体）',
         '2026-10-08 Claude作成（特訓・下書き）。移動とおうぎ形を組み合わせる問題。答えは check.py で計算し直して検算。'
         '【保留】図なしで伝えられ、5章のほかの特訓と重ならない問題は 3 問まで。70問にしていない。')
U.expr('半径 6 cm、中心角 90° のおうぎ形を、その中心のまわりに 90° 回転移動させた。おうぎ形が通過した部分の面積を、π を使って答えよう（cm²）。', '18π',
       '①回転すると、中心角 90 + 90 ＝ 180° のおうぎ形（半円）の部分を通る　②π × 36 × (180/360) ＝ 18π　③答え 18π cm²',
       lambda π: π * 36 * F(180, 360))
U.num('1辺 10 cm の正三角形の1つの角の二等分線をひいた。二等分線と、その角をはさむ辺がつくる角は何度か答えよう。', '30',
      '①正三角形の1つの角は 60°　②二等分して 30°　③答え 30度',
      lambda: F(180, 3) / 2, units=['度', '°'])
U.expr('半径 4 cm の円を、その中心が直線上を 10 cm 動くように平行移動させた。円が通過した部分の面積を、π を使って答えよう（cm²）。', '80+16π',
       '①両はしの半円 2 つで円1つ分：16π　②まん中は、縦 8 cm（直径）、横 10 cm の長方形：80　③80 + 16π。答え 80+16π cm²',
       lambda π: 8 * 10 + π * 16, form='nosign')
