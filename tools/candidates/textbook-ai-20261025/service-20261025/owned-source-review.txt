  const id=await textbookOperationId(request),key='openai-operation:'+uid+':'+id,priorRaw=await db.prepare('SELECT v FROM auth_kv WHERE k=?').bind(key).first('v');
  const authorized=async()=>{
   if(!await active()||!await consent(db,uid,clock()))return false;
   try{const current=await textbookAccess(env,uid,clock());return current.accessVersion===rights.accessVersion&&current.features.chat===true;}catch{return false;}
  };
  const writeGuard=()=>{const at=clock(),g=textbookWriteGuard(rights,{uid,owner,feature:'chat',now:at});return {sql:g.sql+' AND EXISTS(SELECT 1 FROM auth_kv WHERE k=? AND v=? AND (expires_at=0 OR expires_at>?))',values:[...g.values,'openai-access:'+uid,consentRaw,at]};};
  if(!priorRaw){
   if(quota.aiLeft===0||quota.dailyLeft<1)return {error:'ai_plan'};
   const gate=await trialStatus(db,clock());if(gate.stopped)return {error:'ai_stopped',trial:gate};if(gate.used>=gate.limit)return {error:'ai_trial_daily',trial:gate};
   const rawBudget=await db.prepare('SELECT v FROM auth_kv WHERE k=?').bind(monthKey(clock())).first('v');
   if(rawBudget!==null&&(!/^\d{1,9}$/.test(rawBudget)||Number(rawBudget)>=cap))return {error:'ai_budget'};
  }
  const snapshot=await capturedTextbookChat(db,uid,request,{plan:rights.plan,readContext,env,now:clock(),guard:writeGuard,operationExists:!!priorRaw});
  const content=textbookChatContent(snapshot.context,request.text,snapshot.history),policy=requestBody(rights.plan,'chat',content),digest=await sha(JSON.stringify(policy.body));
  if(priorRaw){
   let prior;try{prior=JSON.parse(priorRaw);}catch{return {error:'operation_invalid'};}
   if(prior.digest!==digest)return {error:'id_conflict'};
   if(prior.state==='done'){
    if(!await authorized())return {error:'session_changed'};
    return {real:true,model:policy.body.model,value:textbookChatReply(JSON.stringify(prior.value)),context:snapshot.context,cached:true,creditsUsed:0};
   }
   if(prior.state==='started')return {error:'request_outcome_unknown'};
   if(prior.state==='rejected'){
    if(!await recoverTextbookRefund(db,uid,id))return {error:'refund_pending'};
    quota=legacy?await planStatus(env,uid,clock()):await textbookChatQuota(db,rights,{uid,owner,now:clock()});
   }
  }
  // Frozen source snapshots may be read for recovery, but cannot authorize a higher old tier.
  if(snapshot.plan!==rights.plan)return {error:'plan_changed'};
  if(quota.aiLeft===0||quota.dailyLeft<1)return {error:'ai_plan'};
  return runRealAITrial(db,async signal=>{
   const operation=await reservePricedOperation(db,env,uid,id,digest,policy,clock());if(operation.error)return operation;
   if(operation.state!=='reserved')return {error:'request_outcome_unknown'};
   if(!await startOperation(db,uid,id,operation.token))return {error:'request_in_progress'};
   if(!await authorized()){await rejectUnchargedOperation(db,uid,id,operation.token);return {error:'session_changed'};}
   let quotaReceipt={},issued=false;
   const charged=legacy?await planUse(env,db,uid,clock(),quotaReceipt,2):await reserveTextbookChat(db,rights,{uid,owner,now:clock(),requestId:await sha(id+':'+operation.token),fingerprint:digest});
   if(legacy?!charged:charged.error){await rejectUnchargedOperation(db,uid,id,operation.token);return {error:'ai_plan'};}
   if(!legacy)quotaReceipt=charged.receipt;
   // Persist before any provider call. A lost write/charge acknowledgement stays started.
   if(!await pinTextbookReceipt(db,uid,id,operation.token,quotaReceipt))return {error:'save_failed',retrySafe:false};
   const guardedFetcher=async(...args)=>{
    if(!enabled(env)||!await authorized()||(await trialStatus(db,clock())).stopped)throw Error('session_changed');
    if(args[0]!=='https://api.openai.com/v1/moderations')issued=true;
    return fetcher(...args);
   };
   const result=await generate(env,rights.plan,'chat',content,{fetcher:guardedFetcher,validate:textbookChatReply,signal,reserve:async amount=>amount===policy.reservation});
   if(result.error){
    const safe=!issued||(result.error==='billing_not_ready'&&result.retrySafe===true);
    if(safe){const released=await rejectUnchargedOperation(db,uid,id,operation.token),refunded=released&&await recoverTextbookRefund(db,uid,id);return {...result,retrySafe:!!refunded};}
    // No second dispatch/refund after a lost billable response or invalid output.
    return {...result,retrySafe:false};
   }
   if(!await authorized())return {error:'session_changed',retrySafe:false};
   if(!await completePricedOperation(db,uid,id,operation.token,result))return {error:'save_failed',retrySafe:false};
   if(!await authorized())return {error:'session_changed',retrySafe:false};
   return {...result,context:snapshot.context,cached:false,creditsUsed:1};
  },clock());
 }catch(error){
  const code=error?.message;
  if(['context_changed','context_not_found','id_conflict','plan_changed','release_changed'].includes(code))return {error:code==='release_changed'?'plan_changed':code};
  if(code==='invalid_textbook_chat'||code==='unsupported_chat_input'||code==='private_chat_input')return {error:'ai_unsupported'};
  return {error:'textbook_chat_unavailable'};
 }
}
